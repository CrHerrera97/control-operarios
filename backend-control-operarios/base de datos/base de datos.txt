CREATE DATABASE control_operarios
  CHARACTER SET utf8mb4
  COLLATE utf8mb4_unicode_ci;

USE control_operarios;

CREATE TABLE usuarios (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    usuario VARCHAR(50) NOT NULL UNIQUE,
    contrasena_hash VARCHAR(255) NOT NULL,
    nombre_completo VARCHAR(150) NOT NULL,
    rol ENUM('ADMIN', 'OPERARIO') NOT NULL DEFAULT 'OPERARIO',
    activo BOOLEAN NOT NULL DEFAULT TRUE,
    creado_en DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP
) ENGINE=InnoDB;

CREATE TABLE movimientos (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    usuario_id BIGINT UNSIGNED NOT NULL,
    movimiento ENUM('ENTRADA', 'SALIDA') NOT NULL,
    obra VARCHAR(150) NOT NULL,
    latitud DECIMAL(10, 8) NOT NULL,
    longitud DECIMAL(11, 8) NOT NULL,
    fotografia MEDIUMBLOB NOT NULL,
    ip_address VARCHAR(45) NULL,
    dispositivo_id CHAR(32) NULL,
    fecha_hora DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT fk_movimientos_usuario
        FOREIGN KEY (usuario_id)
        REFERENCES usuarios(id)
        ON UPDATE CASCADE
        ON DELETE RESTRICT,

    INDEX idx_movimientos_fecha (fecha_hora),
    INDEX idx_movimientos_tipo (movimiento),
    INDEX idx_movimientos_usuario (usuario_id),
    INDEX idx_movimientos_dispositivo (dispositivo_id)
) ENGINE=InnoDB;

ALTER TABLE movimientos
    ADD COLUMN notas VARCHAR(255) NULL;

-- Requiere que el servidor MySQL tenga event_scheduler=ON.
SET @zona_horaria_anterior = @@session.time_zone;
SET time_zone = '-05:00';

CREATE EVENT IF NOT EXISTS cierre_diario_movimientos
ON SCHEDULE EVERY 1 DAY
STARTS (TIMESTAMP(DATE_ADD(CURDATE(), INTERVAL IF(CURTIME() < '23:30:00', 0, 1) DAY), '23:30:00'))
ON COMPLETION PRESERVE
ENABLE
DO
        INSERT INTO movimientos
                (usuario_id, movimiento, obra, latitud, longitud, fotografia, ip_address, dispositivo_id, fecha_hora, notas)
        SELECT entrada.usuario_id, 'SALIDA', entrada.obra, entrada.latitud, entrada.longitud,
                     entrada.fotografia, entrada.ip_address, entrada.dispositivo_id, CURRENT_TIMESTAMP,
                     'NO REALIZÓ SALIDA'
        FROM movimientos AS entrada
        WHERE entrada.movimiento = 'ENTRADA'
            AND DATE(entrada.fecha_hora) = CURRENT_DATE()
            AND NOT EXISTS (
                    SELECT 1
                    FROM movimientos AS posterior
                    WHERE posterior.usuario_id = entrada.usuario_id
                        AND (posterior.fecha_hora > entrada.fecha_hora
                                 OR (posterior.fecha_hora = entrada.fecha_hora AND posterior.id > entrada.id))
            );

SET time_zone = @zona_horaria_anterior;
